async function copySignature(mode: CopyMode, html: string, plain: string): Promise<CopyResult> {
  if (!navigator.clipboard) throw new Error("Clipboard unavailable");
  if (mode === "plain") { await navigator.clipboard.writeText(plain); return "plain"; }
  if (mode === "html") { await navigator.clipboard.writeText(html); return "html"; }
  if (navigator.clipboard.write && typeof ClipboardItem !== "undefined") {
    try { await navigator.clipboard.write([new ClipboardItem({ "text/html": new Blob([html], { type: "text/html" }), "text/plain": new Blob([plain], { type: "text/plain" }) })]); return "rich"; }
    catch { await navigator.clipboard.writeText(plain); return "plain"; }
  }
  await navigator.clipboard.writeText(plain);
  return "plain";
}
